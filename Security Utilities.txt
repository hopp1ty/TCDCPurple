LinuxTools

find / -xdev -perm -4000 -type f 2>/dev/null    # SUID (compare to a baseline)
find / -xdev -perm -2000 -type f 2>/dev/null    # SGID
find / -xdev -type f -perm -o+w 2>/dev/null     # world-writable files
find / -xdev -type d -perm -o+w ! -perm -1000 2>/dev/null   # writable dirs w/o sticky bit
getcap -r / 2>/dev/null                         # file capabilities (privesc vector)
sudo -l; getfacl -R /etc 2>/dev/null | grep -B5 "other::.w"
lsattr -R /etc 2>/dev/null | grep -- '-i-'      # immutable flags (can hide malware)
find / -xdev -nouser -o -nogroup 2>/dev/null    # orphaned files

find / -xdev -name ".*" -type f -path "*/tmp/*" 2>/dev/null
getfattr -R -d /home /tmp 2>/dev/null

Baseline workflow

mkdir -p /root/base && cd /root/base
ss -tulpn > ports.txt
ps -eo user,pid,ppid,cmd --sort=pid > procs.txt
find / -xdev -perm -4000 2>/dev/null | sort > suid.txt
systemctl list-unit-files --state=enabled > units.txt
for u in $(cut -d: -f1 /etc/passwd); do echo "== $u"; crontab -u $u -l 2>/dev/null; done > cron.txt
cp /etc/passwd /etc/shadow /etc/sudoers . ; find / -name authorized_keys 2>/dev/null > keys.txt
